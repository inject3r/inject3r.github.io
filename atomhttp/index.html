import asyncio
from atomhttp import AtomHTTP

async def main():
    client = AtomHTTP({
        'baseURL': 'https://jsonplaceholder.typicode.com',
        'timeout': 10
    })
    
    response = await client.get('/posts/1')
    print(f"Status: {response.status}")
    print(f"Title: {response.data['title']}")

asyncio.run(main())
